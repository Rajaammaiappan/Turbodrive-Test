    # -------------------------------------------------------------- #
    #  Going headless AFTER a manual sign-in
    # -------------------------------------------------------------- #
    def hide_window(self) -> None:
        """
        Minimise the browser window.

        Chromium cannot be switched from headed to headless while running -
        --headless=new is a launch flag - so this is the safe way to get the
        window out of the way: the session, cookies and downloads are all
        untouched, the window is simply no longer on screen.
        """
        try:
            wid = self.browser_ws.call("Browser.getWindowForTarget",
                                       {"targetId": self._main_target_id})
            self.browser_ws.call("Browser.setWindowBounds", {
                "windowId": wid["windowId"],
                "bounds": {"windowState": "minimized"}})
            self.status("Browser minimised - it keeps working in the "
                        "background.")
        except Exception as exc:                 # noqa: BLE001
            self.log.debug("Could not minimise the window: %s", exc)

    def go_headless(self) -> bool:
        """
        Restart the browser headless, reusing the signed-in profile.

        The sign-in cookies live in the profile directory, so a headless
        relaunch on the SAME profile is already authenticated and no second
        MFA prompt appears.

        This only works with "persistent_profile": true. With a throw-away
        profile the session dies with the old process, and the headless
        window would stop at an MFA prompt nobody can see or approve - so
        that case minimises instead and says why.

        Returns True if it really went headless.
        """
        if getattr(self, "_temp_profile", False):
            self.status(
                "Not switching to headless: 'persistent_profile' is false, so "
                "the signed-in session would be lost and the headless browser "
                "would stop at an MFA prompt you cannot approve. Minimising "
                "instead - set 'persistent_profile': true to enable this.",
                logging.WARNING)
            self.hide_window()
            return False

        if self.headless:
            return True                          # already headless

        url = getattr(self, "_last_url", "")
        download_dir = self.download_dir
        self.status("Switching to headless mode (reusing your session)...")
        try:
            self.quit()
        except Exception as exc:                 # noqa: BLE001
            self.log.debug("Error closing the visible browser: %s", exc)

        self.headless = True
        self._refreshed_url = None               # allow the one-off refresh again
        try:
            self.start_browser()
        except Exception as exc:                 # noqa: BLE001
            self.status(f"Headless relaunch failed ({exc}).", logging.ERROR)
            raise

        self.set_download_dir(download_dir)
        if url:
            self.open_site(url)
        self.status("Now running headless.")
        return True
